library(stringr)

url_2010_2023 <- "https://www.imdb.com/search/title/?title_type=feature&release_date=2010-01-01,2023-12-30&num_votes=2500,&country_of_origin=TR&count=250"
url_until_2010 <- "https://www.imdb.com/search/title/?title_type=feature&release_date=,2009-12-31&num_votes=2500,&country_of_origin=TR&count=250"

page1 <- read_html(url_2010_2023)
page2 <- read_html(url_until_2010)


titles_2010_2023 <- page1 |> html_nodes(".ipc-title__text") |> html_text()
titles_2010_2023 <- tail(head(titles_2010_2023,-1),-1)
titles_2010_2023 <- gsub("^\\d+\\.\\s*", "", titles_2010_2023)


titles_until_2010 <- page2 |> html_nodes(".ipc-title__text") |> html_text()
titles_until_2010 <- page2 |> html_nodes(".ipc-title__text") |> html_text()

votes_2010_2023 <- page1 |> html_nodes(".ipc-rating-star--voteCount") |> html_text()

votes_2010_2023 <- gsub("[^0-9]", "", votes_2010_2023)

votes_until_2010[votes_until_2010 >= 1] <- votes_until_2010[votes_until_2010 >= 1] * 1000


durations_2010_2023 <- page1 |> html_nodes(".dli-title-metadata-item:nth-child(2)") |> html_text()
durations_until_2010 <- page2 |> html_nodes(".dli-title-metadata-item:nth-child(2)") |> html_text()


convert_to_minutes <- function(duration_text) {
  durations <- str_match(duration_text, "(\\d+)h\\s?(\\d*)m?") 
  hours <- as.numeric(durations[, 2])
  minutes <- as.numeric(durations[, 3])
  
  total_minutes <- (ifelse(is.na(hours), 0, hours) * 60) + ifelse(is.na(minutes), 0, minutes)
  
}


durations_2010_2023_minutes <- convert_to_minutes(durations_2010_2023)
durations_until_2010_minutes <- convert_to_minutes(durations_until_2010)


ratings_2010_2023 <- page1 |> html_nodes(".ratingGroup--imdb-rating") |> html_text()
ratings_until_2010 <- page2 |> html_nodes(".ratingGroup--imdb-rating") |> html_text()
